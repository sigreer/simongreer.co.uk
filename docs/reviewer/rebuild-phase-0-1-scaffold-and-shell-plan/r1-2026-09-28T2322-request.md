   91	// scripts/gen-route-manifest.mjs
   92	// Reads the OLD content tree (Astro 5 layout) and emits docs/specs/route-manifest.json.
   93	// Run once from the repo root while src/content still holds the legacy frontmatter.
   94	import { readdirSync, readFileSync, writeFileSync } from 'node:fs';
   95	import { join, basename } from 'node:path';
   96	
   97	const root = process.cwd();
   98	const collections = [
   99	  { name: 'blog', dir: 'src/content/blog', prefix: '/blog/' },
  100	  { name: 'tech', dir: 'src/content/tech', prefix: '/tech/' },
  101	  { name: 'projects', dir: 'src/content/projects', prefix: '/tech/projects/' },
  102	];
  103	
  104	function frontmatter(file) {
  105	  const text = readFileSync(file, 'utf8');
  106	  const m = text.match(/^---\n([\s\S]*?)\n---/);
  107	  if (!m) throw new Error(`no frontmatter in ${file}`);
  108	  const out = {};
  109	  for (const line of m[1].split('\n')) {
  110	    const kv = line.match(/^(\w+):\s*(.*)$/);
  111	    if (kv) out[kv[1]] = kv[2].replace(/^['"]|['"]$/g, '').trim();
  112	  }
  113	  return out;
  114	}
  115	
  116	const rows = [];
  117	const seen = new Map(); // newPath -> file, to detect collisions
  118	
  119	for (const c of collections) {
  120	  for (const f of readdirSync(join(root, c.dir)).filter((n) => n.endsWith('.mdx') || n.endsWith('.md'))) {
  121	    const file = join(c.dir, f);
  122	    const fm = frontmatter(join(root, file));
  123	    const filename = basename(f).replace(/\.mdx?$/, '');
  124	    const slug = fm.slug || filename;               // Astro 5 glob loader used slug as id when present
  125	    const currentPath = `${c.prefix}${slug}/`;
  126	    const status = fm.status || 'published';
  127	
  128	    if (status !== 'published') {
  129	      rows.push({ collection: c.name, file, currentPath, newPath: null, action: 'drop', reason: `status ${status}` });
  130	      continue;
  131	    }
  132	    if (fm.title === 'Project Title') {
  133	      rows.push({ collection: c.name, file, currentPath, newPath: null, action: 'drop', reason: 'placeholder entry' });
  134	      continue;
  135	    }
  136	
  137	    let newId = slug;
  138	    let action = 'keep';
  139	    let reason = slug === filename ? 'unchanged' : `rename file ${filename}.mdx -> ${slug}.mdx`;
  140	
  141	    if (slug.endsWith('.mdx')) {                      // voip project has ".mdx" inside its slug
  142	      newId = slug.replace(/\.mdx$/, '');
  143	      action = 'redirect';
  144	      reason = 'slug contained .mdx';
  145	    }
  146	    const newPath = `${c.prefix}${newId}/`;
  147	
  148	    if (seen.has(newPath)) {                          // collision: filename that equals the slug wins
  149	      const other = seen.get(newPath);
  150	      const winner = basename(other.file).replace(/\.mdx?$/, '') === newId ? other : null;
  151	      if (winner) {
  152	        rows.push({ collection: c.name, file, currentPath, newPath: `${c.prefix}${filename}/`, action: 'keep',
  153	          reason: `slug collided with ${other.file}; uses filename ${filename}` });
  154	        continue;
  155	      }
  156	    }
  157	    seen.set(newPath, { file });
  158	    rows.push({ collection: c.name, file, currentPath, newPath, action, reason });
  159	  }
  160	}
  161	
  162	// Static aliases and pages
  163	rows.push({ collection: 'static', file: null, currentPath: '/me/', newPath: '/me/personally/', action: 'redirect', reason: 'existing rule' });
  164	rows.push({ collection: 'static', file: null, currentPath: '/me/get-in-touch/', newPath: '/get-in-touch/', action: 'redirect', reason: 'replaces meta-refresh page' });
  165	rows.push({ collection: 'static', file: null, currentPath: '/home/', newPath: null, action: 'drop', reason: 'orphan duplicate of /' });
  166	for (const [oldPage, target] of [
  167	  ['vpn-setup-routing-wireguard-ipsec-openvpn', 'networking-and-security'],
  168	  ['system-administration', 'system-design-and-deployment'],
  169	]) {
  170	  rows.push({ collection: 'services', file: null, currentPath: `/hire-me/${oldPage}/`, newPath: `/hire-me/${target}/`, action: 'redirect', reason: 'page merged (spec §4.3)' });
  171	}
  172	for (const p of ['web-development','business-apps','cloud-and-hosted','networking-and-security','storage-and-nas','software-development','data-and-databases','system-design-and-deployment']) {
  173	  rows.push({ collection: 'services', file: null, currentPath: `/hire-me/${p}/`, newPath: `/hire-me/${p}/`, action: 'keep', reason: 'service page' });
  174	}
  175	rows.push({ collection: 'services', file: null, currentPath: null, newPath: '/hire-me/ai-and-automation/', action: 'keep', reason: 'new service page (spec §4.3)' });
  176	
  177	rows.sort((a, b) => `${a.collection}${a.currentPath}`.localeCompare(`${b.collection}${b.currentPath}`));
  178	writeFileSync('docs/specs/route-manifest.json', JSON.stringify({ generatedAt: new Date().toISOString(), rows }, null, 2) + '\n');
  179	const counts = rows.reduce((acc, r) => ((acc[r.action] = (acc[r.action] || 0) + 1), acc), {});
  180	console.log(JSON.stringify(counts));
  181	```

   33	### 3.1 Repository strategy
   34

   38	
   39	```
   40	Browser ──> Cloudflare edge (zone simongreer.co.uk)
   41	              ├─ static asset match ──> Workers Static Assets (HTML, _astro/*, pagefind/*, images)
   42	              └─ no match ───────────> Worker (Astro server entry)
